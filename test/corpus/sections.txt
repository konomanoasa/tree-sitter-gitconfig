==================
Header layout preserves names, delimiters and later variables
==================
[ core ] x=y
[remote "x" ]
[remote.origin 	]
z=w
---

(document
  (section
    header: (section_header
      opening: (section_open)
      name: (section_name (name_text))
      closing: (section_close))
    (variable
      name: (variable_name (name_text))
      operator: (assignment_operator)
      value: (value (value_text))
      (line_ending)))
  (section
    header: (section_header
      opening: (section_open)
      name: (section_name (name_text))
      subsection: (subsection
        opening: (quote_open)
        (subsection_text)
        closing: (quote_close))
      closing: (section_close))
    (line_ending))
  (section
    header: (section_header
      opening: (section_open)
      name: (section_name (name_text))
      subsection: (subsection
        (subsection_separator)
        (subsection_text))
      closing: (section_close))
    (line_ending)
    (variable
      name: (variable_name (name_text))
      operator: (assignment_operator)
      value: (value (value_text)))))
